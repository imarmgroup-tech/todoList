SQL Table Schema Script (สำหรับสร้างตารางใน Supabase)

นำโค้ด SQL ด้านล่างนี้ไปรันในช่อง **SQL Editor** บน Dashboard ของ Supabase เพื่อสร้างตาราง todos พร้อมเปิดใช้งาน Realtime: 

-- 1. Create todos table
CREATE TABLE IF NOT EXISTS public.todos (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    title TEXT NOT NULL,
    priority TEXT DEFAULT 'medium' CHECK (priority IN ('high', 'medium', 'low')),
    category TEXT DEFAULT 'General',
    due_time TEXT,
    completed BOOLEAN DEFAULT FALSE,
    source TEXT DEFAULT 'manual',
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Enable Row Level Security (RLS) & Public Policies for Demo
ALTER TABLE public.todos ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow public read access" ON public.todos FOR SELECT USING (true);
CREATE POLICY "Allow public insert access" ON public.todos FOR INSERT WITH CHECK (true);
CREATE POLICY "Allow public update access" ON public.todos FOR UPDATE USING (true);
CREATE POLICY "Allow public delete access" ON public.todos FOR DELETE USING (true);

-- 3. Enable Realtime Replication for table todos
ALTER PUBLICATION supabase_realtime ADD TABLE public.todos;

คู่มือ Telegram Webhook & Supabase Edge Function

คุณสามารถสร้าง **Supabase Edge Function** เพื่อรับ Webhook จาก Telegram Bot และบันทึกงานลงใน Supabase โดยอัตโนมัติ ซึ่งจะกระตุ้น Realtime Sync บนหน้าเว็บทันที

// โค้ดสำหรับ Supabase Edge Function (Deno/TypeScript)
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2'

Deno.serve(async (req) => {
  const payload = await req.json()
  const text = payload.message?.text || ''

  if (text.startsWith('/add')) {
    const title = text.replace('/add', '').trim()
    const supabase = createClient(
      Deno.env.get('SUPABASE_URL')!,
      Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
    )

    await supabase.from('todos').insert([{
      title: title || 'งานใหม่จาก Telegram',
      priority: text.includes('#high') ? 'high' : 'medium',
      source: 'telegram'
    }])
  }

  return new Response('OK', { status: 200 })
})
